    96→function maskPassword(pw: string): string {
    97→  if (!pw) return ''
    98→  return '•'.repeat(Math.min(pw.length, 16))
    99→}

   101→function parseCustomFields(raw: string | null): string[] {
   102→  if (!raw) return []
   103→  const parsed = safeParse<string[]>(raw, [])
   104→  return Array.isArray(parsed) ? parsed.filter((s) => typeof s === 'string') : []
   105→}
   106→
   107→function sourceCustomFields(source: WordPressSource): string[] {
   108→  return parseCustomFields(source.customFields)
   109→}

   111→/** Custom fields chip editor: text input + add button + removable chips. */
   112→function CustomFieldsEditor({
   113→  fields,
   114→  onChange,
   115→}: {
   116→  fields: string[]
   117→  onChange: (next: string[]) => void
   118→}) {
   119→  const [draft, setDraft] = React.useState('')
   120→
   121→  function addField() {
   122→    const value = draft.trim()
   123→    if (!value) return
   124→    if (fields.includes(value)) {
   125→      setDraft('')
   126→      return
   127→    }
   128→    onChange([...fields, value])
   129→    setDraft('')
   130→  }
   131→
   132→  function removeField(field: string) {
   133→    onChange(fields.filter((f) => f !== field))
   134→  }
   135→
   136→  return (
   137→    <div className="grid gap-2">
   138→      <div className="flex gap-2">
   139→        <Input
   140→          value={draft}
   141→          onChange={(e) => setDraft(e.target.value)}
   142→          onKeyDown={(e) => {
   143→            if (e.key === 'Enter') {
   144→              e.preventDefault()
   145→              addField()
   146→            }
   147→          }}
   148→          placeholder="например, _price, _sku, _stock_status"
   149→        />
   150→        <Button type="button" variant="outline" onClick={addField}>
   151→          <Plus className="size-4" />
   152→          Добавить
   153→        </Button>
   154→      </div>
   155→      {fields.length > 0 ? (
   156→        <div className="flex flex-wrap gap-1.5">
   157→          {fields.map((f) => (
   158→            <Badge
   159→              key={f}
   160→              variant="secondary"
   161→              className="gap-1 font-mono text-xs"
   162→            >
   163→              {f}
   164→              <button
   165→                type="button"
   166→                className="ml-0.5 rounded-sm hover:bg-background/80"
   167→                onClick={() => removeField(f)}
   168→                aria-label={`Удалить ${f}`}
   169→              >
   170→                <X className="size-3" />
   171→              </button>
   172→            </Badge>
   173→          ))}
   174→        </div>
   175→      ) : (
   176→        <p className="text-xs text-muted-foreground">
   177→          Список пуст. Можно добавить ключи кастомных полей WordPress (meta),
   178→          которые нужно подтягивать из товаров.
   179→        </p>
   180→      )}
   181→    </div>
   182→  )
   183→}

   369→function ProductList({
   370→  sourceId,
   371→  customFields,
   372→}: {
   373→  sourceId: string
   374→  customFields: string[]
   375→}) {
   376→  const { data, isLoading, isError, refetch, error } = useQuery<ProductsResponse>({
   377→    queryKey: ['wordpress', sourceId, 'products'],
   378→    queryFn: async () => {
   379→      const res = await fetch(`/api/wordpress/${sourceId}/products`)
   380→      if (!res.ok) throw new Error('Не удалось загрузить товары')
   381→      return (await res.json()) as ProductsResponse
   382→    },
   383→  })
   384→
   385→  if (isLoading) {
   386→    return (
   387→      <div className="grid gap-2">
   388→        {Array.from({ length: 4 }).map((_, i) => (
   389→          <Skeleton key={i} className="h-24 w-full" />
   390→        ))}
   391→      </div>
   392→    )
   393→  }
   394→
   395→  if (isError) {
   396→    return (
   397→      <Alert variant="destructive">
   398→        <AlertTitle>Ошибка запроса</AlertTitle>
   399→        <AlertDescription>
   400→          {error instanceof Error ? error.message : 'Неизвестная ошибка'}.{' '}
   401→          <Button
   402→            variant="link"
   403→            className="h-auto p-0 text-destructive"
   404→            onClick={() => refetch()}
   405→          >
   406→            Повторить
   407→          </Button>
   408→        </AlertDescription>
   409→      </Alert>
   410→    )
   411→  }
   412→
   413→  if (data?.error) {
   414→    return (
   415→      <Alert variant="destructive">
   416→        <AlertTitle>WordPress вернул ошибку</AlertTitle>
   417→        <AlertDescription className="break-words">{data.error}</AlertDescription>
   418→      </Alert>
   419→    )
   420→  }
   421→
   422→  const products = data?.products ?? []
   423→  if (products.length === 0) {
   424→    return (
   425→      <div className="rounded-md border border-dashed p-8 text-center text-sm text-muted-foreground">
   426→        Товары не найдены. Проверьте URL, тип записи и учётные данные.
   427→      </div>
   428→    )
   429→  }
   430→
   431→  return (
   432→    <div className="grid gap-2">
   433→      <div className="text-xs text-muted-foreground">
   434→        Найдено товаров: {products.length}
   435→      </div>
   436→      <ScrollArea className="max-h-96 rounded-md border">
   437→        <div className="grid gap-2 p-2">
   438→          {products.map((p) => {
   439→            const metaEntries = Object.entries(p.meta ?? {}).slice(0, 4)
   440→            const filteredMeta =
   441→              customFields.length > 0
   442→                ? metaEntries.filter(([k]) => customFields.includes(k))
   443→                : metaEntries
   444→            return (
   445→              <div
   446→                key={p.id}
   447→                className="flex gap-3 rounded-md border bg-card p-3 text-sm"
   448→              >
   449→                <div className="size-12 shrink-0 overflow-hidden rounded bg-muted">
   450→                  {p.featuredImage ? (
   451→                    <img
   452→                      src={p.featuredImage}
   453→                      alt={p.title}
   454→                      className="size-full object-cover"
   455→                      loading="lazy"
   456→                    />
   457→                  ) : (
   458→                    <div className="flex size-full items-center justify-center text-muted-foreground">
   459→                      <ImageOff className="size-4" />
   460→                    </div>
   461→                  )}
   462→                </div>
   463→                <div className="min-w-0 flex-1">
   464→                  <div className="flex items-center justify-between gap-2">
   465→                    <a
   466→                      href={p.link}
   467→                      target="_blank"
   468→                      rel="noopener noreferrer"
   469→                      className="truncate font-medium text-foreground hover:underline"
   470→                    >
   471→                      {p.title || 'Без названия'}
   472→                    </a>
   473→                    <ExternalLink className="size-3 shrink-0 text-muted-foreground" />
   474→                  </div>
   475→                  <div className="truncate font-mono text-xs text-muted-foreground">
   476→                    /{p.slug || p.id}
   477→                  </div>
   478→                  {p.excerpt && (
   479→                    <div className="mt-1 line-clamp-2 text-xs text-muted-foreground">
   480→                      {p.excerpt}
   481→                    </div>
   482→                  )}
   483→                  {filteredMeta.length > 0 && (
   484→                    <div className="mt-1.5 flex flex-wrap gap-1">
   485→                      {filteredMeta.map(([k, v]) => (
   486→                        <Badge
   487→                          key={k}
   488→                          variant="secondary"
   489→                          className="font-mono text-[10px]"
   490→                        >
   491→                          <span className="text-muted-foreground">{k}:</span>
   492→                          <span className="ml-1 max-w-[140px] truncate">{v}</span>
   493→                        </Badge>
   494→                      ))}
   495→                    </div>
   496→                  )}
   497→                </div>
   498→              </div>
   499→            )
   500→          })}
   501→        </div>
   502→      </ScrollArea>
   503→    </div>
   504→  )
   505→}

   664→  const [dialogOpen, setDialogOpen] = React.useState(false)

   666→  const [deleteTarget, setDeleteTarget] = React.useState<WordPressSource | null>(null)
   667→  const [productsSource, setProductsSource] = React.useState<WordPressSource | null>(
   668→    null,
   669→  )
   670→
   671→  const createMutation = useMutation({
   672→    mutationFn: async (values: SourceFormValues) => {
   673→      const res = await fetch('/api/wordpress', {
   674→        method: 'POST',
   675→        headers: { 'Content-Type': 'application/json' },
   676→        body: JSON.stringify({
   677→          name: values.name,
   678→          siteUrl: values.siteUrl,
   679→          username: values.username,
   680→          appPassword: values.appPassword,
   681→          postType: values.postType,
   682→          customFields: values.customFields,
   683→        }),
   684→      })
   685→      if (!res.ok) {
   686→        const data = await res.json().catch(() => ({}))
   687→        throw new Error(data?.error || 'Ошибка создания')
   688→      }
   689→      return res.json()
   690→    },
   691→    onSuccess: () => {
   692→      toast.success('Источник добавлен')
   693→      queryClient.invalidateQueries({ queryKey: ['wordpress'] })
   694→      setDialogOpen(false)
   695→    },
   696→    onError: (e: Error) => toast.error(e.message),
   697→  })
   698→
   699→  const updateMutation = useMutation({
   700→    mutationFn: async (vars: { id: string; values: SourceFormValues }) => {
   701→      const res = await fetch(`/api/wordpress/${vars.id}`, {
   702→        method: 'PUT',
   703→        headers: { 'Content-Type': 'application/json' },
   704→        body: JSON.stringify({
   705→          name: vars.values.name,
   706→          siteUrl: vars.values.siteUrl,
   707→          username: vars.values.username,
   708→          appPassword: vars.values.appPassword,
   709→          postType: vars.values.postType,
   710→          customFields: vars.values.customFields,
   711→        }),
   712→      })
   713→      if (!res.ok) {
   714→        const data = await res.json().catch(() => ({}))
   715→        throw new Error(data?.error || 'Ошибка обновления')
   716→      }
   717→      return res.json()
   718→    },
   719→    onSuccess: () => {
   720→      toast.success('Источник обновлён')
   721→      queryClient.invalidateQueries({ queryKey: ['wordpress'] })
   722→      setDialogOpen(false)
   723→      setEditTarget(null)
   724→    },
   725→    onError: (e: Error) => toast.error(e.message),
   726→  })
   727→
   728→  const toggleMutation = useMutation({
   729→    mutationFn: async (vars: { id: string; active: boolean }) => {
   730→      const res = await fetch(`/api/wordpress/${vars.id}`, {
   731→        method: 'PUT',
   732→        headers: { 'Content-Type': 'application/json' },
   733→        body: JSON.stringify({ active: vars.active }),
   734→      })
   735→      if (!res.ok) throw new Error('Не удалось изменить статус')
   736→      return res.json()
   737→    },
   738→    onSuccess: () => {
   739→      queryClient.invalidateQueries({ queryKey: ['wordpress'] })
   740→    },
   741→    onError: (e: Error) => toast.error(e.message),
   742→  })
   743→
   744→  const deleteMutation = useMutation({
   745→    mutationFn: async (id: string) => {
   746→      const res = await fetch(`/api/wordpress/${id}`, { method: 'DELETE' })
   747→      if (!res.ok) throw new Error('Не удалось удалить')
   748→      return res.json()
   749→    },
   750→    onSuccess: () => {
   751→      toast.success('Источник удалён')
   752→      queryClient.invalidateQueries({ queryKey: ['wordpress'] })
   753→      setDeleteTarget(null)
   754→    },
   755→    onError: (e: Error) => toast.error(e.message),
   756→  })
   757→
   758→  const submitting = createMutation.isPending || updateMutation.isPending
   759→

   762→    setDialogOpen(true)

   764→
   765→  function openEdit(src: WordPressSource) {

   767→    setDialogOpen(true)
   768→  }
   769→
   770→  function handleSubmit(values: SourceFormValues) {

   772→      updateMutation.mutate({ id: editTarget.id, values })
   773→    } else {
   774→      createMutation.mutate(values)
   775→    }
   776→  }

   780→      <Card className="bg-teal-50/40 dark:bg-teal-950/10">

   783→          <CardDescription>
   784→            Подключите ваш WordPress-магазин. Посты будут создаваться на основе
   785→            товаров и их кастомных полей. (Полная интеграция с кастомными полями —
   786→            следующий этап.)
   787→          </CardDescription>
   788→        </CardHeader>
   789→      </Card>
   790→
   791→      <div className="flex items-center justify-between gap-2">
   792→        <div className="text-sm text-muted-foreground">
   793→          Источников:{' '}
   794→          <span className="font-medium text-foreground">{data?.length ?? 0}</span>
   795→        </div>
   796→        <Button
   797→          onClick={openCreate}
   798→          className="bg-teal-700 text-white hover:bg-teal-700/90"
   799→        >

   801→          Добавить источник
   802→        </Button>

   805→      {isLoading ? (

   822→          </CardContent>

   825→        <Card className="border-dashed">
   826→          <CardContent className="flex flex-col items-center gap-3 py-12 text-center">
   827→            <div className="rounded-full bg-muted p-3">
   828→              <Globe className="size-5 text-muted-foreground" />
   829→            </div>
   830→            <div>
   831→              <div className="font-medium">Нет подключённых источников</div>

   833→                Добавьте WordPress-магазин, чтобы подтягивать товары для постов.

   835→            </div>
   836→            <Button

   841→              Добавить источник
