   134→   134→function pad(n: number): string {
   135→   135→  return n < 10 ? `0${n}` : String(n)
   136→   136→}
   137→   137→
   138→   138→function localDateKey(d: Date): string {
   139→   139→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
   140→   140→}
   141→   141→
   142→   142→/** Stable hash from a string — used to derive deterministic-looking stats. */
   143→   143→function hashStr(s: string): number {
   144→   144→  let h = 0
   145→   145→  for (let i = 0; i < s.length; i++) {
   146→   146→    h = (h * 31 + s.charCodeAt(i)) | 0
   147→   147→  }
   148→   148→  return Math.abs(h)
   149→   149→}
   150→   150→
   151→   151→/** 1234 → "1,2K", 836 → "836", 12000 → "12,0K" */
   152→   152→function formatCount(n: number): string {
   153→   153→  if (n >= 1000) {
   154→   154→    const k = n / 1000
   155→   155→    return `${k.toFixed(1).replace('.', ',')}K`
   156→   156→  }
   157→   157→  return String(n)
   158→   158→}

   164→   164→export function DashboardSection() {
   165→   165→  const [filter, setFilter] = React.useState<FeedFilter>('all')
   166→   166→
   167→   167→  const postsQuery = useQuery<PostListItem[]>({
   168→   168→    queryKey: ['posts', 'dashboard-feed'],
   169→   169→    queryFn: async () => {
   170→   170→      const res = await fetch('/api/posts')
   171→   171→      if (!res.ok) throw new Error('Не удалось загрузить посты')
   172→   172→      return res.json() as Promise<PostListItem[]>
   173→   173→    },
   174→   174→    staleTime: 30_000,
   175→   175→  })
   176→   176→
   177→   177→  const appendicesQuery = useQuery<AppendixTemplate[]>({
   178→   178→    queryKey: ['appendices', 'dashboard-feed'],
   179→   179→    queryFn: async () => {
   180→   180→      const res = await fetch('/api/appendices')
   181→   181→      if (!res.ok) throw new Error('Не удалось загрузить приписки')
   182→   182→      return res.json() as Promise<AppendixTemplate[]>
   183→   183→    },
   184→   184→    staleTime: 30_000,
   185→   185→  })
   186→   186→
   187→   187→  const appendixMap = React.useMemo(() => {
   188→   188→    const m = new Map<string, AppendixTemplate>()
   189→   189→    for (const a of appendicesQuery.data ?? []) m.set(a.id, a)
   190→   190→    return m
   191→   191→  }, [appendicesQuery.data])
   192→   192→
   193→   193→  // API returns DESC by scheduledAt — keep latest first.
   194→   194→  const sortedPosts = React.useMemo(() => {
   195→   195→    if (!postsQuery.data) return []
   196→   196→    return [...postsQuery.data].sort(
   197→   197→      (a, b) =>
   198→   198→        new Date(b.scheduledAt).getTime() - new Date(a.scheduledAt).getTime(),
   199→   199→    )
   200→   200→  }, [postsQuery.data])
   201→   201→
   202→   202→  const filteredPosts = React.useMemo(() => {
   203→   203→    const now = new Date()
   204→   204→    return sortedPosts.filter((p) => {
   205→   205→      if (filter === 'all') return true
   206→   206→      if (filter === 'scheduled')
   207→   207→        return (
   208→   208→          p.status === 'scheduled' ||
   209→   209→          p.status === 'publishing' ||
   210→   210→          p.status === 'draft'
   211→   211→        )
   212→   212→      if (filter === 'published') return p.status === 'published'
   213→   213→      if (filter === 'today') return isSameDay(parseISO(p.scheduledAt), now)
   214→   214→      return true
   215→   215→    })
   216→   216→  }, [sortedPosts, filter])
   217→   217→
   218→   218→  const loading = postsQuery.isLoading || appendicesQuery.isLoading
   219→   219→
   220→   220→  return (
   221→   221→    <div className="grid gap-6 lg:grid-cols-[minmax(0,1fr)_300px]">
   222→   222→      {/* CENTER: feed */}
   223→   223→      <section className="flex min-w-0 flex-col gap-4">
   224→   224→        <FeedHeader
   225→   225→          count={filteredPosts.length}
   226→   226→          filter={filter}
   227→   227→          onFilterChange={setFilter}
   228→   228→        />
   229→   229→        {loading ? (
   230→   230→          <div className="space-y-4">
   231→   231→            {Array.from({ length: 4 }).map((_, i) => (
   232→   232→              <FeedCardSkeleton key={i} />
   233→   233→            ))}
   234→   234→          </div>
   235→   235→        ) : filteredPosts.length === 0 ? (
   236→   236→          <FeedEmptyState filter={filter} />
   237→   237→        ) : (
   238→   238→          <div className="space-y-4">
   239→   239→            {filteredPosts.map((post) => (
   240→   240→              <FeedPostCard
   241→   241→                key={post.id}
   242→   242→                post={post}
   243→   243→                appendixMap={appendixMap}
   244→   244→              />
   245→   245→            ))}
   246→   246→          </div>
   247→   247→        )}
   248→   248→      </section>
   249→   249→
   250→   250→      {/* RIGHT: mini-calendar + quick stats */}
   251→   251→      <aside className="flex flex-col gap-4 lg:sticky lg:top-[72px] lg:self-start">
   252→   252→        <MiniCalendar />
   253→   253→        <QuickStats />
   254→   254→      </aside>
   255→   255→    </div>
   256→   256→  )
   257→   257→}

   302→   302→function FeedPostCard({
   303→   303→  post,
   304→   304→  appendixMap,
   305→   305→}: {
   306→   306→  post: PostListItem
   307→   307→  appendixMap: Map<string, AppendixTemplate>
   308→   308→}) {
   309→   309→  const setSelectedPostId = useStore((s) => s.setSelectedPostId)
   310→   310→  const setSection = useStore((s) => s.setSection)
   311→   311→
   312→   312→  const date = parseISO(post.scheduledAt)
   313→   313→  const statusMeta = POST_STATUS_META[post.status] ?? POST_STATUS_META.scheduled
   314→   314→
   315→   315→  // Primary target = first target's account (default to telegram if none)
   316→   316→  const primaryTarget = post.targets[0] ?? null
   317→   317→  const primaryPlatform: Platform = primaryTarget?.account.platform ?? 'telegram'
   318→   318→  const primaryAccountName = primaryTarget?.account.name
   319→   319→
   320→   320→  // Appendix attached to the primary target
   321→   321→  const appendix =
   322→   322→    primaryTarget?.appendixId != null
   323→   323→      ? (appendixMap.get(primaryTarget.appendixId) ?? undefined)
   324→   324→      : undefined
   325→   325→  const appendixPosition: AppendixPosition | undefined =
   326→   326→    appendix?.position === 'prepend'
   327→   327→      ? 'prepend'
   328→   328→      : appendix?.position === 'append'
   329→   329→        ? 'append'
   330→   330→        : undefined
   331→   331→
   332→   332→  const composedText = composePostText(
   333→   333→    post.content ?? '',
   334→   334→    appendix?.body,
   335→   335→    appendixPosition,
   336→   336→  )
   337→   337→  const photos = safeParse<string[]>(post.mediaUrls, [])
   338→   338→
   339→   339→  // Unique platforms across all targets → colored dots
   340→   340→  const platforms = Array.from(
   341→   341→    new Set(
   342→   342→      post.targets
   343→   343→        .map((t) => t.account.platform)
   344→   344→        .filter(
   345→   345→          (p): p is Platform => p === 'telegram' || p === 'max' || p === 'vk',
   346→   346→        ),
   347→   347→    ),
   348→   348→  )
   349→   349→
   350→   350→  // Realistic-looking engagement metrics — only for published posts.
   351→   351→  const isPublished = post.status === 'published'
   352→   352→  const baseHash = hashStr(post.id)
   353→   353→  const views = isPublished
   354→   354→    ? formatCount(120 + (baseHash % 4800))
   355→   355→    : undefined
   356→   356→  const reactions = isPublished
   357→   357→    ? formatCount(8 + (hashStr(post.id + 'r') % 180))
   358→   358→    : undefined
   359→   359→  const comments =
   360→   360→    isPublished && primaryPlatform === 'vk'
   361→   361→      ? formatCount(1 + (hashStr(post.id + 'c') % 40))
   362→   362→      : undefined
   363→   363→  const shares =
   364→   364→    isPublished && primaryPlatform === 'vk'
   365→   365→      ? formatCount(1 + (hashStr(post.id + 's') % 25))
   366→   366→      : undefined
   367→   367→
   368→   368→  const time = format(date, 'HH:mm')
   369→   369→
   370→   370→  const handleOpen = () => {
   371→   371→    setSelectedPostId(post.id)
   372→   372→    setSection('posts')
   373→   373→  }
   374→   374→
   375→   375→  return (
   376→   376→    <Card className="gap-0 overflow-hidden rounded-xl p-0 shadow-sm">
   377→   377→      {/* Meta strip */}
   378→   378→      <div className="flex items-center justify-between gap-2 px-4 pt-3">
   379→   379→        <div className="flex min-w-0 items-center gap-2">
   380→   380→          <Clock className="size-3.5 shrink-0 text-muted-foreground" />
   381→   381→          <span className="truncate text-sm font-medium tabular-nums">
   382→   382→            {format(date, 'd MMM, HH:mm', { locale: ru })}
   383→   383→          </span>
   384→   384→          <span className="hidden truncate text-xs text-muted-foreground sm:inline">
   385→   385→            {formatDistanceToNow(date, { locale: ru, addSuffix: true })}
   386→   386→          </span>
   387→   387→        </div>
   388→   388→        <div className="flex shrink-0 items-center gap-1.5">
   389→   389→          <Badge
   390→   390→            variant="outline"
   391→   391→            className={cn('border-transparent', statusMeta.bg, statusMeta.text)}
   392→   392→          >
   393→   393→            {statusMeta.label}
   394→   394→          </Badge>
   395→   395→          <div className="flex items-center gap-0.5">
   396→   396→            {platforms.map((p) => {
   397→   397→              const meta = PLATFORMS[p]
   398→   398→              if (!meta) return null
   399→   399→              return (
   400→   400→                <span
   401→   401→                  key={p}
   402→   402→                  className="inline-block size-2 rounded-full"
   403→   403→                  style={{ backgroundColor: meta.hex }}
   404→   404→                  title={meta.label}
   405→   405→                />
   406→   406→              )
   407→   407→            })}
   408→   408→          </div>
   409→   409→        </div>
   410→   410→      </div>
   411→   411→
   412→   412→      {/* Preview body */}
   413→   413→      <div className="px-4 pb-4 pt-2">
   414→   414→        <SocialPostPreview
   415→   415→          platform={primaryPlatform}
   416→   416→          text={composedText}
   417→   417→          photos={photos}
   418→   418→          accountName={primaryAccountName}
   419→   419→          time={time}
   420→   420→          views={views}
   421→   421→          reactions={reactions}
   422→   422→          comments={comments}
   423→   423→          shares={shares}
   424→   424→        />
   425→   425→      </div>
   426→   426→
   427→   427→      {/* Footer strip */}
   428→   428→      <div className="flex items-center gap-2 border-t bg-muted/30 px-4 py-2">
   429→   429→        <span className="truncate text-xs text-muted-foreground">
   430→   430→          {post.title || 'Без названия'}
   431→   431→        </span>
   432→   432→        <div className="ml-auto flex items-center gap-1.5">
   433→   433→          {post.wordpressRef ? (
   434→   434→            <Badge
   435→   435→              variant="outline"
   436→   436→              className="gap-1 border-transparent bg-teal-50 text-[10px] text-teal-700"
   437→   437→            >
   438→   438→              <Newspaper className="size-2.5" />
   439→   439→              WP
   440→   440→            </Badge>
   441→   441→          ) : null}
   442→   442→          {photos.length > 0 ? (
   443→   443→            <Badge
   444→   444→              variant="outline"
   445→   445→              className="gap-1 border-transparent bg-slate-100 text-[10px] text-slate-600"
   446→   446→            >
   447→   447→              <ImageIcon className="size-2.5" />
   448→   448→              {photos.length}
   449→   449→            </Badge>
   450→   450→          ) : null}
   451→   451→          <Button
   452→   452→            variant="ghost"
   453→   453→            size="sm"
   454→   454→            className="h-7 gap-1 px-2 text-teal-700"
   455→   455→            onClick={handleOpen}
   456→   456→          >
   457→   457→            Открыть
   458→   458→            <ArrowRight className="size-3" />
   459→   459→          </Button>
   460→   460→        </div>
   461→   461→      </div>
   462→   462→    </Card>
   463→   463→  )
   464→   464→}

   650→   650→function MiniDayCell({

   664→   664→  const hasPublished = inMonth && !!bucket && bucket.published > 0

   671→   671→      className={cn(

   673→   673→        !inMonth && 'text-muted-foreground/40',
