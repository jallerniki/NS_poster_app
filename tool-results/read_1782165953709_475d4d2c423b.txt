   151→/**
   152→ * Render a URL template with placeholders replaced by example values — used for
   153→ * the live preview so the user can see what the final URL will look like.
   154→ */
   155→function renderUrlTemplate(template: string): string {
   156→  if (!template) return ''
   157→  return template
   158→    .replace(/\{\{sku\}\}/g, '13876')
   159→    .replace(/\{\{id\}\}/g, '13876')
   160→    .replace(/\{\{slug\}\}/g, 'ljon-s-hlopkom-peserico')
   161→    .replace(/\{\{category\}\}/g, 'Лён')
   162→    .replace(/\{\{categorySlug\}\}/g, 'len')
   163→}

   584→
   585→      <CardFooter className="gap-2">
   586→        <Button size="sm" variant="outline" onClick={onEdit}>
   587→          <Pencil className="size-3.5" />
   588→          Редактировать
   589→        </Button>
   590→        <Button
   591→          size="sm"
   592→          variant="ghost"
   593→          className="text-destructive hover:text-destructive"
   594→          onClick={onDelete}
   595→        >
   596→          <Trash2 className="size-3.5" />
   597→          Удалить
   598→        </Button>
   599→      </CardFooter>
   600→    </Card>
   601→  )
   602→}
   603→
   604→export function ButtonsSection() {
   605→  const queryClient = useQueryClient()
   606→
   607→  const { data, isLoading, isError, refetch } = useQuery<ButtonPreset[]>({
   608→    queryKey: ['button-presets'],
   609→    queryFn: async () => {
   610→      const res = await fetch('/api/button-presets')
   611→      if (!res.ok) throw new Error('Не удалось загрузить пресеты кнопок')
   612→      return (await res.json()) as ButtonPreset[]
   613→    },
   614→  })
   615→
   616→  const [dialogOpen, setDialogOpen] = React.useState(false)
   617→  const [editTarget, setEditTarget] = React.useState<ButtonPreset | null>(null)
   618→  const [deleteTarget, setDeleteTarget] = React.useState<ButtonPreset | null>(
   619→    null,
   620→  )
   621→
   622→  const createMutation = useMutation({
   623→    mutationFn: async (values: FormValues) => {
   624→      const res = await fetch('/api/button-presets', {
   625→        method: 'POST',
   626→        headers: { 'Content-Type': 'application/json' },
   627→        body: JSON.stringify(values),
   628→      })
   629→      if (!res.ok) {
   630→        const j = await res.json().catch(() => ({}))
   631→        throw new Error(j?.error || 'Ошибка создания')
   632→      }
   633→      return res.json()
   634→    },
   635→    onSuccess: () => {
   636→      toast.success('Пресет создан')
   637→      queryClient.invalidateQueries({ queryKey: ['button-presets'] })
   638→      setDialogOpen(false)
   639→    },
   640→    onError: (e: Error) => toast.error(e.message),
   641→  })
   642→
   643→  const updateMutation = useMutation({
   644→    mutationFn: async (vars: { id: string; values: FormValues }) => {
   645→      const res = await fetch(`/api/button-presets/${vars.id}`, {
   646→        method: 'PUT',
   647→        headers: { 'Content-Type': 'application/json' },
   648→        body: JSON.stringify(vars.values),
   649→      })
   650→      if (!res.ok) {
   651→        const j = await res.json().catch(() => ({}))
   652→        throw new Error(j?.error || 'Ошибка обновления')
   653→      }
   654→      return res.json()
   655→    },
   656→    onSuccess: () => {
   657→      toast.success('Пресет обновлён')
   658→      queryClient.invalidateQueries({ queryKey: ['button-presets'] })
   659→      setDialogOpen(false)
   660→      setEditTarget(null)
   661→    },
   662→    onError: (e: Error) => toast.error(e.message),
   663→  })
   664→
   665→  const deleteMutation = useMutation({
   666→    mutationFn: async (id: string) => {
   667→      const res = await fetch(`/api/button-presets/${id}`, {
   668→        method: 'DELETE',
   669→      })
   670→      if (!res.ok) throw new Error('Не удалось удалить')
   671→      return res.json()
   672→    },
   673→    onSuccess: () => {
   674→      toast.success('Пресет удалён')
   675→      queryClient.invalidateQueries({ queryKey: ['button-presets'] })
   676→      setDeleteTarget(null)
   677→    },
   678→    onError: (e: Error) => toast.error(e.message),
   679→  })
   680→
   681→  const submitting = createMutation.isPending || updateMutation.isPending
   682→
   683→  function openCreate() {
   684→    setEditTarget(null)
   685→    setDialogOpen(true)
   686→  }
   687→
   688→  function openEdit(preset: ButtonPreset) {
   689→    setEditTarget(preset)
   690→    setDialogOpen(true)
   691→  }
   692→
   693→  function handleSubmit(values: FormValues) {
   694→    if (editTarget) {
   695→      updateMutation.mutate({ id: editTarget.id, values })
   696→    } else {
   697→      createMutation.mutate(values)
   698→    }
   699→  }
   700→
   701→  return (
   702→    <div className="grid gap-4">
   703→      <Card className="bg-teal-50/40 dark:bg-teal-950/10">
   704→        <CardHeader>
   705→          <CardTitle className="flex items-center gap-2 text-base">
   706→            <MousePointerClick className="size-4 text-teal-700 dark:text-teal-300" />
   707→            Кнопки в постах
   708→          </CardTitle>
   709→          <CardDescription>
   710→            Пресет кнопок описывает две inline-кнопки — «Товар» и «Категория» —
   711→            которые автоматически добавляются к сообщению при создании поста из
   712→            карточки товара. URL-шаблоны содержат переменные вида{' '}
   713→            <code className="rounded bg-muted px-1 py-0.5 text-[11px]">
   714→              {'{{sku}}'}
   715→            </code>
   716→            , которые подставляются из данных товара WordPress. Создайте по
   717→            одному пресету для каждой площадки (Telegram и MAX).
   718→          </CardDescription>
   719→        </CardHeader>
   720→      </Card>
   721→
   722→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   723→        <div className="text-sm text-muted-foreground">
   724→          {data && data.length > 0
   725→            ? `Всего пресетов: ${data.length}`
   726→            : 'Создайте первый пресет кнопок'}
   727→        </div>
   728→        <Button
   729→          onClick={openCreate}
   730→          className="bg-teal-700 text-white hover:bg-teal-700/90"
   731→        >
   732→          <Plus className="size-4" />
   733→          Новый пресет
   734→        </Button>
   735→      </div>
   736→
   737→      {isLoading ? (
   738→        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   739→          {Array.from({ length: 2 }).map((_, i) => (
   740→            <Skeleton key={i} className="h-56 w-full" />
   741→          ))}
   742→        </div>
   743→      ) : isError ? (
   744→        <Card>
   745→          <CardContent className="py-6 text-sm text-destructive">
   746→            Не удалось загрузить пресеты кнопок.{' '}
   747→            <Button
   748→              variant="link"
   749→              className="h-auto p-0"
   750→              onClick={() => refetch()}
   751→            >
   752→              Повторить
   753→            </Button>
   754→          </CardContent>
   755→        </Card>
   756→      ) : !data || data.length === 0 ? (
   757→        <Card className="border-dashed">
   758→          <CardContent className="flex flex-col items-center gap-3 py-12 text-center">
   759→            <div className="rounded-full bg-muted p-3">
   760→              <MousePointerClick className="size-5 text-muted-foreground" />
   761→            </div>
   762→            <div>
   763→              <div className="font-medium">Пресетов кнопок пока нет</div>
   764→              <div className="text-sm text-muted-foreground">
   765→                Создайте первый пресет — например, кнопки для Telegram-бота
   766→                магазина.
   767→              </div>
   768→            </div>
   769→            <Button
   770→              onClick={openCreate}
   771→              className="bg-teal-700 text-white hover:bg-teal-700/90"
   772→            >
   773→              <Plus className="size-4" />
   774→              Новый пресет
   775→            </Button>
   776→          </CardContent>
   777→        </Card>
   778→      ) : (
   779→        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   780→          {data.map((preset) => (
   781→            <ButtonPresetCard
   782→              key={preset.id}
   783→              preset={preset}
   784→              onEdit={() => openEdit(preset)}
   785→              onDelete={() => setDeleteTarget(preset)}
   786→            />
   787→          ))}
   788→        </div>
   789→      )}
   790→
   791→      <ButtonPresetFormDialog
   792→        open={dialogOpen}
   793→        onOpenChange={(o) => {
   794→          setDialogOpen(o)
   795→          if (!o) setEditTarget(null)
   796→        }}
   797→        mode={editTarget ? 'edit' : 'create'}
   798→        initial={editTarget}
   799→        onSubmit={handleSubmit}
   800→        submitting={submitting}
   801→      />
   802→
   803→      <AlertDialog
   804→        open={!!deleteTarget}
   805→        onOpenChange={(o) => !o && setDeleteTarget(null)}
   806→      >
   807→        <AlertDialogContent>
   808→          <AlertDialogHeader>
   809→            <AlertDialogTitle>Удалить пресет кнопок?</AlertDialogTitle>
   810→            <AlertDialogDescription>
   811→              {deleteTarget?.name ? (
   812→                <>
   813→                  Пресет «{deleteTarget.name}» будет удалён без возможности
   814→                  восстановления. Кнопки перестанут добавляться к новым постам
   815→                  этой площадки.
   816→                </>
   817→              ) : (
   818→                'Это действие нельзя отменить.'
   819→              )}
   820→            </AlertDialogDescription>
   821→          </AlertDialogHeader>
   822→          <AlertDialogFooter>
   823→            <AlertDialogCancel>Отмена</AlertDialogCancel>
   824→            <AlertDialogAction
   825→              className="bg-destructive text-white hover:bg-destructive/90"
   826→              disabled={deleteMutation.isPending}
   827→              onClick={(e) => {
   828→                e.preventDefault()
   829→                if (deleteTarget) deleteMutation.mutate(deleteTarget.id)
   830→              }}
   831→            >
   832→              {deleteMutation.isPending ? 'Удаление…' : 'Удалить'}
   833→            </AlertDialogAction>
   834→          </AlertDialogFooter>
   835→        </AlertDialogContent>
   836→      </AlertDialog>
   837→    </div>
   838→  )
   839→}
   840→
   841→export default ButtonsSection
