   134→function pad(n: number): string {
   135→  return n < 10 ? `0${n}` : String(n)
   136→}
   137→
   138→function localDateKey(d: Date): string {
   139→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
   140→}
   141→
   142→/** Stable hash from a string — used to derive deterministic-looking stats. */
   143→function hashStr(s: string): number {
   144→  let h = 0
   145→  for (let i = 0; i < s.length; i++) {
   146→    h = (h * 31 + s.charCodeAt(i)) | 0
   147→  }
   148→  return Math.abs(h)
   149→}
   150→
   151→/** 1234 → "1,2K", 836 → "836", 12000 → "12,0K" */
   152→function formatCount(n: number): string {
   153→  if (n >= 1000) {
   154→    const k = n / 1000
   155→    return `${k.toFixed(1).replace('.', ',')}K`
   156→  }
   157→  return String(n)
   158→}

   164→export function DashboardSection() {
   165→  const [filter, setFilter] = React.useState<FeedFilter>('all')
   166→
   167→  const postsQuery = useQuery<PostListItem[]>({
   168→    queryKey: ['posts', 'dashboard-feed'],
   169→    queryFn: async () => {
   170→      const res = await fetch('/api/posts')
   171→      if (!res.ok) throw new Error('Не удалось загрузить посты')
   172→      return res.json() as Promise<PostListItem[]>
   173→    },
   174→    staleTime: 30_000,
   175→  })
   176→
   177→  const appendicesQuery = useQuery<AppendixTemplate[]>({
   178→    queryKey: ['appendices', 'dashboard-feed'],
   179→    queryFn: async () => {
   180→      const res = await fetch('/api/appendices')
   181→      if (!res.ok) throw new Error('Не удалось загрузить приписки')
   182→      return res.json() as Promise<AppendixTemplate[]>
   183→    },
   184→    staleTime: 30_000,
   185→  })
   186→
   187→  const appendixMap = React.useMemo(() => {
   188→    const m = new Map<string, AppendixTemplate>()
   189→    for (const a of appendicesQuery.data ?? []) m.set(a.id, a)
   190→    return m
   191→  }, [appendicesQuery.data])
   192→
   193→  // API returns DESC by scheduledAt — keep latest first.
   194→  const sortedPosts = React.useMemo(() => {
   195→    if (!postsQuery.data) return []
   196→    return [...postsQuery.data].sort(
   197→      (a, b) =>
   198→        new Date(b.scheduledAt).getTime() - new Date(a.scheduledAt).getTime(),
   199→    )
   200→  }, [postsQuery.data])
   201→
   202→  const filteredPosts = React.useMemo(() => {
   203→    const now = new Date()
   204→    return sortedPosts.filter((p) => {
   205→      if (filter === 'all') return true
   206→      if (filter === 'scheduled')
   207→        return (
   208→          p.status === 'scheduled' ||
   209→          p.status === 'publishing' ||
   210→          p.status === 'draft'
   211→        )
   212→      if (filter === 'published') return p.status === 'published'
   213→      if (filter === 'today') return isSameDay(parseISO(p.scheduledAt), now)
   214→      return true
   215→    })
   216→  }, [sortedPosts, filter])
   217→
   218→  const loading = postsQuery.isLoading || appendicesQuery.isLoading
   219→
   220→  return (
   221→    <div className="grid gap-6 lg:grid-cols-[minmax(0,1fr)_300px]">
   222→      {/* CENTER: feed */}
   223→      <section className="flex min-w-0 flex-col gap-4">
   224→        <FeedHeader
   225→          count={filteredPosts.length}
   226→          filter={filter}
   227→          onFilterChange={setFilter}
   228→        />
   229→        {loading ? (
   230→          <div className="space-y-4">
   231→            {Array.from({ length: 4 }).map((_, i) => (
   232→              <FeedCardSkeleton key={i} />
   233→            ))}
   234→          </div>
   235→        ) : filteredPosts.length === 0 ? (
   236→          <FeedEmptyState filter={filter} />
   237→        ) : (
   238→          <div className="space-y-4">
   239→            {filteredPosts.map((post) => (
   240→              <FeedPostCard
   241→                key={post.id}
   242→                post={post}
   243→                appendixMap={appendixMap}
   244→              />
   245→            ))}
   246→          </div>
   247→        )}
   248→      </section>
   249→
   250→      {/* RIGHT: mini-calendar + quick stats */}
   251→      <aside className="flex flex-col gap-4 lg:sticky lg:top-[72px] lg:self-start">
   252→        <MiniCalendar />
   253→        <QuickStats />
   254→      </aside>
   255→    </div>
   256→  )
   257→}

   302→function FeedPostCard({
   303→  post,
   304→  appendixMap,
   305→}: {
   306→  post: PostListItem
   307→  appendixMap: Map<string, AppendixTemplate>
   308→}) {
   309→  const setSelectedPostId = useStore((s) => s.setSelectedPostId)
   310→  const setSection = useStore((s) => s.setSection)
   311→
   312→  const date = parseISO(post.scheduledAt)
   313→  const statusMeta = POST_STATUS_META[post.status] ?? POST_STATUS_META.scheduled
   314→
   315→  // Primary target = first target's account (default to telegram if none)
   316→  const primaryTarget = post.targets[0] ?? null
   317→  const primaryPlatform: Platform = primaryTarget?.account.platform ?? 'telegram'
   318→  const primaryAccountName = primaryTarget?.account.name
   319→
   320→  // Appendix attached to the primary target
   321→  const appendix =
   322→    primaryTarget?.appendixId != null
   323→      ? (appendixMap.get(primaryTarget.appendixId) ?? undefined)
   324→      : undefined
   325→  const appendixPosition: AppendixPosition | undefined =
   326→    appendix?.position === 'prepend'
   327→      ? 'prepend'
   328→      : appendix?.position === 'append'
   329→        ? 'append'
   330→        : undefined
   331→
   332→  const composedText = composePostText(
   333→    post.content ?? '',
   334→    appendix?.body,
   335→    appendixPosition,
   336→  )
   337→  const photos = safeParse<string[]>(post.mediaUrls, [])
   338→
   339→  // Unique platforms across all targets → colored dots
   340→  const platforms = Array.from(
   341→    new Set(
   342→      post.targets
   343→        .map((t) => t.account.platform)
   344→        .filter(
   345→          (p): p is Platform => p === 'telegram' || p === 'max' || p === 'vk',
   346→        ),
   347→    ),
   348→  )
   349→
   350→  // Realistic-looking engagement metrics — only for published posts.
   351→  const isPublished = post.status === 'published'
   352→  const baseHash = hashStr(post.id)
   353→  const views = isPublished
   354→    ? formatCount(120 + (baseHash % 4800))
   355→    : undefined
   356→  const reactions = isPublished
   357→    ? formatCount(8 + (hashStr(post.id + 'r') % 180))
   358→    : undefined
   359→  const comments =
   360→    isPublished && primaryPlatform === 'vk'
   361→      ? formatCount(1 + (hashStr(post.id + 'c') % 40))
   362→      : undefined
   363→  const shares =
   364→    isPublished && primaryPlatform === 'vk'
   365→      ? formatCount(1 + (hashStr(post.id + 's') % 25))
   366→      : undefined
   367→
   368→  const time = format(date, 'HH:mm')
   369→
   370→  const handleOpen = () => {
   371→    setSelectedPostId(post.id)
   372→    setSection('posts')
   373→  }
   374→
   375→  return (
   376→    <Card className="gap-0 overflow-hidden rounded-xl p-0 shadow-sm">
   377→      {/* Meta strip */}
   378→      <div className="flex items-center justify-between gap-2 px-4 pt-3">
   379→        <div className="flex min-w-0 items-center gap-2">
   380→          <Clock className="size-3.5 shrink-0 text-muted-foreground" />
   381→          <span className="truncate text-sm font-medium tabular-nums">
   382→            {format(date, 'd MMM, HH:mm', { locale: ru })}
   383→          </span>
   384→          <span className="hidden truncate text-xs text-muted-foreground sm:inline">
   385→            {formatDistanceToNow(date, { locale: ru, addSuffix: true })}
   386→          </span>
   387→        </div>
   388→        <div className="flex shrink-0 items-center gap-1.5">
   389→          <Badge
   390→            variant="outline"
   391→            className={cn('border-transparent', statusMeta.bg, statusMeta.text)}
   392→          >
   393→            {statusMeta.label}
   394→          </Badge>
   395→          <div className="flex items-center gap-0.5">
   396→            {platforms.map((p) => {
   397→              const meta = PLATFORMS[p]
   398→              if (!meta) return null
   399→              return (
   400→                <span
   401→                  key={p}
   402→                  className="inline-block size-2 rounded-full"
   403→                  style={{ backgroundColor: meta.hex }}
   404→                  title={meta.label}
   405→                />
   406→              )
   407→            })}
   408→          </div>
   409→        </div>
   410→      </div>
   411→
   412→      {/* Preview body */}
   413→      <div className="px-4 pb-4 pt-2">
   414→        <SocialPostPreview
   415→          platform={primaryPlatform}
   416→          text={composedText}
   417→          photos={photos}
   418→          accountName={primaryAccountName}
   419→          time={time}
   420→          views={views}
   421→          reactions={reactions}
   422→          comments={comments}
   423→          shares={shares}
   424→        />
   425→      </div>
   426→
   427→      {/* Footer strip */}
   428→      <div className="flex items-center gap-2 border-t bg-muted/30 px-4 py-2">
   429→        <span className="truncate text-xs text-muted-foreground">
   430→          {post.title || 'Без названия'}
   431→        </span>
   432→        <div className="ml-auto flex items-center gap-1.5">
   433→          {post.wordpressRef ? (
   434→            <Badge
   435→              variant="outline"
   436→              className="gap-1 border-transparent bg-teal-50 text-[10px] text-teal-700"
   437→            >
   438→              <Newspaper className="size-2.5" />
   439→              WP
   440→            </Badge>
   441→          ) : null}
   442→          {photos.length > 0 ? (
   443→            <Badge
   444→              variant="outline"
   445→              className="gap-1 border-transparent bg-slate-100 text-[10px] text-slate-600"
   446→            >
   447→              <ImageIcon className="size-2.5" />
   448→              {photos.length}
   449→            </Badge>
   450→          ) : null}
   451→          <Button
   452→            variant="ghost"
   453→            size="sm"
   454→            className="h-7 gap-1 px-2 text-teal-700"
   455→            onClick={handleOpen}
   456→          >
   457→            Открыть
   458→            <ArrowRight className="size-3" />
   459→          </Button>
   460→        </div>
   461→      </div>
   462→    </Card>
   463→  )
   464→}

   526→function MiniCalendar() {
   527→  const setSection = useStore((s) => s.setSection)
   528→  const today = React.useMemo(() => new Date(), [])
   529→  const [viewDate, setViewDate] = React.useState<Date>(() =>
   530→    startOfMonth(today),
   531→  )
   532→
   533→  const year = viewDate.getFullYear()
   534→  const month = viewDate.getMonth() + 1
   535→
   536→  const calendarQuery = useQuery<CalendarDay[]>({
   537→    queryKey: ['calendar', year, month],
   538→    queryFn: async () => {
   539→      const params = new URLSearchParams({
   540→        year: String(year),
   541→        month: String(month),
   542→      })
   543→      const res = await fetch(`/api/calendar?${params.toString()}`)
   544→      if (!res.ok) throw new Error('Не удалось загрузить календарь')
   545→      return res.json() as Promise<CalendarDay[]>
   546→    },
   547→    staleTime: 30_000,
   548→  })
   549→
   550→  const cells = React.useMemo(() => {
   551→    const monthStart = startOfMonth(viewDate)
   552→    const monthEnd = endOfMonth(viewDate)
   553→    const gridStart = startOfWeek(monthStart, { weekStartsOn: 1 })
   554→    const gridEnd = endOfWeek(monthEnd, { weekStartsOn: 1 })
   555→    return eachDayOfInterval({ start: gridStart, end: gridEnd })
   556→  }, [viewDate])
   557→
   558→  const bucketsByKey = React.useMemo(() => {
   559→    const map = new Map<string, CalendarDay>()
   560→    for (const b of calendarQuery.data ?? []) {
   561→      map.set(b.date, b)
   562→    }
   563→    return map
   564→  }, [calendarQuery.data])
   565→
   566→  return (
   567→    <Card className="gap-0 p-3">
   568→      {/* Header */}
   569→      <div className="flex items-center justify-between gap-1 px-1 pb-2">
   570→        <div className="text-sm font-semibold capitalize">
   571→          {format(viewDate, 'LLLL yyyy', { locale: ru })}
   572→        </div>
   573→        <div className="flex items-center gap-0.5">
   574→          <Button
   575→            variant="ghost"
   576→            size="icon"
   577→            className="size-7"
   578→            onClick={() => setViewDate((d) => addMonths(d, -1))}
   579→            aria-label="Предыдущий месяц"
   580→          >
   581→            <ChevronLeft className="size-4" />
   582→          </Button>
   583→          <Button
   584→            variant="ghost"
   585→            size="icon"
   586→            className="size-7"
   587→            onClick={() => setViewDate((d) => addMonths(d, 1))}
   588→            aria-label="Следующий месяц"
   589→          >
   590→            <ChevronRight className="size-4" />
   591→          </Button>
   592→        </div>
   593→      </div>
   594→
   595→      {/* Weekday headers */}
   596→      <div className="grid grid-cols-7 gap-0.5 pb-1">
   597→        {WEEKDAY_HEADERS.map((wd) => (
   598→          <div
   599→            key={wd}
   600→            className="text-center text-[10px] font-medium uppercase text-muted-foreground"
   601→          >
   602→            {wd}
   603→          </div>
   604→        ))}
   605→      </div>
   606→
   607→      {/* Day cells */}
   608→      {calendarQuery.isLoading ? (
   609→        <div className="grid grid-cols-7 gap-0.5">
   610→          {Array.from({ length: 35 }).map((_, i) => (
   611→            <Skeleton key={i} className="h-8 rounded" />
   612→          ))}
   613→        </div>
   614→      ) : (
   615→        <div className="grid grid-cols-7 gap-0.5">
   616→          {cells.map((day) => {
   617→            const key = localDateKey(day)
   618→            const bucket = bucketsByKey.get(key) ?? null
   619→            const inMonth = day.getMonth() === viewDate.getMonth()
   620→            const isToday = isDateToday(day)
   621→            return (
   622→              <MiniDayCell
   623→                key={key}
   624→                date={day}
   625→                bucket={bucket}
   626→                inMonth={inMonth}
   627→                isToday={isToday}
   628→                onClick={() => setSection('calendar')}
   629→              />
   630→            )
   631→          })}
   632→        </div>
   633→      )}
   634→
   635→      {/* Legend */}
   636→      <div className="mt-2 flex items-center justify-center gap-3 text-[10px] text-muted-foreground">
   637→        <div className="flex items-center gap-1">
   638→          <span className="inline-block size-1.5 rounded-full bg-amber-400" />
   639→          запл.
   640→        </div>
   641→        <div className="flex items-center gap-1">
   642→          <span className="inline-block size-1.5 rounded-full bg-emerald-500" />
   643→          опуб.
   644→        </div>
   645→      </div>
   646→    </Card>
   647→  )
   648→}

   650→function MiniDayCell({
   651→  date,
   652→  bucket,
   653→  inMonth,
   654→  isToday,
   655→  onClick,
   656→}: {
   657→  date: Date
   658→  bucket: CalendarDay | null
   659→  inMonth: boolean
   660→  isToday: boolean
   661→  onClick: () => void
   662→}) {
   663→  const hasScheduled = inMonth && !!bucket && bucket.scheduled > 0
   664→  const hasPublished = inMonth && !!bucket && bucket.published > 0
   665→  const hasActivity = hasScheduled || hasPublished
   666→
   667→  return (
   668→    <button
   669→      type="button"
   670→      onClick={onClick}
   671→      className={cn(
   672→        'relative flex h-8 flex-col items-center justify-center rounded transition-colors',
   673→        !inMonth && 'text-muted-foreground/40',
   674→        inMonth && 'text-foreground hover:bg-accent',
   675→        isToday && 'bg-teal-600 text-white hover:bg-teal-600',
   676→      )}
   677→    >
   678→      <span className="text-[11px] font-medium leading-none tabular-nums">
   679→        {format(date, 'd')}
   680→      </span>
   681→      {hasActivity ? (
   682→        <div className="mt-0.5 flex items-center gap-0.5">
   683→          {hasScheduled ? (
   684→            <span className="inline-block size-1 rounded-full bg-amber-400" />
   685→          ) : null}
   686→          {hasPublished ? (
   687→            <span className="inline-block size-1 rounded-full bg-emerald-500" />
   688→          ) : null}
   689→        </div>
   690→      ) : null}
   691→    </button>
   692→  )
   693→}
