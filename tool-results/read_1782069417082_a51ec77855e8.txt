    96→function maskPassword(pw: string): string {
    97→  if (!pw) return ''
    98→  return '•'.repeat(Math.min(pw.length, 16))
    99→}

   101→function parseCustomFields(raw: string | null): string[] {
   102→  if (!raw) return []
   103→  const parsed = safeParse<string[]>(raw, [])
   104→  return Array.isArray(parsed) ? parsed.filter((s) => typeof s === 'string') : []
   105→}
   106→
   107→function sourceCustomFields(source: WordPressSource): string[] {
   108→  return parseCustomFields(source.customFields)
   109→}

   111→/** Custom fields chip editor: text input + add button + removable chips. */
   112→function CustomFieldsEditor({
   113→  fields,
   114→  onChange,
   115→}: {
   116→  fields: string[]
   117→  onChange: (next: string[]) => void
   118→}) {
   119→  const [draft, setDraft] = React.useState('')
   120→
   121→  function addField() {
   122→    const value = draft.trim()
   123→    if (!value) return
   124→    if (fields.includes(value)) {
   125→      setDraft('')
   126→      return
   127→    }
   128→    onChange([...fields, value])
   129→    setDraft('')
   130→  }
   131→
   132→  function removeField(field: string) {
   133→    onChange(fields.filter((f) => f !== field))
   134→  }
   135→
   136→  return (
   137→    <div className="grid gap-2">
   138→      <div className="flex gap-2">
   139→        <Input
   140→          value={draft}
   141→          onChange={(e) => setDraft(e.target.value)}
   142→          onKeyDown={(e) => {
   143→            if (e.key === 'Enter') {
   144→              e.preventDefault()
   145→              addField()
   146→            }
   147→          }}
   148→          placeholder="например, _price, _sku, _stock_status"
   149→        />
   150→        <Button type="button" variant="outline" onClick={addField}>
   151→          <Plus className="size-4" />
   152→          Добавить
   153→        </Button>
   154→      </div>
   155→      {fields.length > 0 ? (
   156→        <div className="flex flex-wrap gap-1.5">
   157→          {fields.map((f) => (
   158→            <Badge
   159→              key={f}
   160→              variant="secondary"
   161→              className="gap-1 font-mono text-xs"
   162→            >
   163→              {f}
   164→              <button
   165→                type="button"
   166→                className="ml-0.5 rounded-sm hover:bg-background/80"
   167→                onClick={() => removeField(f)}
   168→                aria-label={`Удалить ${f}`}
   169→              >
   170→                <X className="size-3" />
   171→              </button>
   172→            </Badge>
   173→          ))}
   174→        </div>
   175→      ) : (
   176→        <p className="text-xs text-muted-foreground">
   177→          Список пуст. Можно добавить ключи кастомных полей WordPress (meta),
   178→          которые нужно подтягивать из товаров.
   179→        </p>
   180→      )}
   181→    </div>
   182→  )
   183→}

   369→function ProductList({
   370→  sourceId,
   371→  customFields,
   372→}: {
   373→  sourceId: string
   374→  customFields: string[]
   375→}) {
   376→  const { data, isLoading, isError, refetch, error } = useQuery<ProductsResponse>({
   377→    queryKey: ['wordpress', sourceId, 'products'],
   378→    queryFn: async () => {
   379→      const res = await fetch(`/api/wordpress/${sourceId}/products`)
   380→      if (!res.ok) throw new Error('Не удалось загрузить товары')
   381→      return (await res.json()) as ProductsResponse
   382→    },
   383→  })
   384→
   385→  if (isLoading) {
   386→    return (
   387→      <div className="grid gap-2">
   388→        {Array.from({ length: 4 }).map((_, i) => (
   389→          <Skeleton key={i} className="h-24 w-full" />
   390→        ))}
   391→      </div>
   392→    )
   393→  }
   394→
   395→  if (isError) {
   396→    return (
   397→      <Alert variant="destructive">
   398→        <AlertTitle>Ошибка запроса</AlertTitle>
   399→        <AlertDescription>
   400→          {error instanceof Error ? error.message : 'Неизвестная ошибка'}.{' '}
   401→          <Button
   402→            variant="link"
   403→            className="h-auto p-0 text-destructive"
   404→            onClick={() => refetch()}
   405→          >
   406→            Повторить
   407→          </Button>
   408→        </AlertDescription>
   409→      </Alert>
   410→    )
   411→  }
   412→
   413→  if (data?.error) {
   414→    return (
   415→      <Alert variant="destructive">
   416→        <AlertTitle>WordPress вернул ошибку</AlertTitle>
   417→        <AlertDescription className="break-words">{data.error}</AlertDescription>
   418→      </Alert>
   419→    )
   420→  }
   421→
   422→  const products = data?.products ?? []
   423→  if (products.length === 0) {
   424→    return (
   425→      <div className="rounded-md border border-dashed p-8 text-center text-sm text-muted-foreground">
   426→        Товары не найдены. Проверьте URL, тип записи и учётные данные.
   427→      </div>
   428→    )
   429→  }
   430→
   431→  return (
   432→    <div className="grid gap-2">
   433→      <div className="text-xs text-muted-foreground">
   434→        Найдено товаров: {products.length}
   435→      </div>
   436→      <ScrollArea className="max-h-96 rounded-md border">
   437→        <div className="grid gap-2 p-2">
   438→          {products.map((p) => {
   439→            const metaEntries = Object.entries(p.meta ?? {}).slice(0, 4)
   440→            const filteredMeta =
   441→              customFields.length > 0
   442→                ? metaEntries.filter(([k]) => customFields.includes(k))
   443→                : metaEntries
   444→            return (
   445→              <div
   446→                key={p.id}
   447→                className="flex gap-3 rounded-md border bg-card p-3 text-sm"
   448→              >
   449→                <div className="size-12 shrink-0 overflow-hidden rounded bg-muted">
   450→                  {p.featuredImage ? (
   451→                    <img
   452→                      src={p.featuredImage}
   453→                      alt={p.title}
   454→                      className="size-full object-cover"
   455→                      loading="lazy"
   456→                    />
   457→                  ) : (
   458→                    <div className="flex size-full items-center justify-center text-muted-foreground">
   459→                      <ImageOff className="size-4" />
   460→                    </div>
   461→                  )}
   462→                </div>
   463→                <div className="min-w-0 flex-1">
   464→                  <div className="flex items-center justify-between gap-2">
   465→                    <a
   466→                      href={p.link}
   467→                      target="_blank"
   468→                      rel="noopener noreferrer"
   469→                      className="truncate font-medium text-foreground hover:underline"
   470→                    >
   471→                      {p.title || 'Без названия'}
   472→                    </a>
   473→                    <ExternalLink className="size-3 shrink-0 text-muted-foreground" />
   474→                  </div>
   475→                  <div className="truncate font-mono text-xs text-muted-foreground">
   476→                    /{p.slug || p.id}
   477→                  </div>
   478→                  {p.excerpt && (
   479→                    <div className="mt-1 line-clamp-2 text-xs text-muted-foreground">
   480→                      {p.excerpt}
   481→                    </div>
   482→                  )}
   483→                  {filteredMeta.length > 0 && (
   484→                    <div className="mt-1.5 flex flex-wrap gap-1">
   485→                      {filteredMeta.map(([k, v]) => (
   486→                        <Badge
   487→                          key={k}
   488→                          variant="secondary"
   489→                          className="font-mono text-[10px]"
   490→                        >
   491→                          <span className="text-muted-foreground">{k}:</span>
   492→                          <span className="ml-1 max-w-[140px] truncate">{v}</span>
   493→                        </Badge>
   494→                      ))}
   495→                    </div>
   496→                  )}
   497→                </div>
   498→              </div>
   499→            )
   500→          })}
   501→        </div>
   502→      </ScrollArea>
   503→    </div>
   504→  )
   505→}

   641→    queryKey: ['wordpress'],
   642→    queryFn: async () => {
   643→      const res = await fetch('/api/wordpress')
   644→      if (!res.ok) throw new Error('Не удалось загрузить источники')
   645→      return (await res.json()) as WordPressSource[]
   646→    },
   647→  })
   648→
   649→  const [dialogOpen, setDialogOpen] = React.useState(false)
   650→  const [editTarget, setEditTarget] = React.useState<WordPressSource | null>(null)
   651→  const [deleteTarget, setDeleteTarget] = React.useState<WordPressSource | null>(null)
   652→  const [productsSource, setProductsSource] = React.useState<WordPressSource | null>(
   653→    null,
   654→  )
   655→
   656→  const createMutation = useMutation({
   657→    mutationFn: async (values: SourceFormValues) => {
   658→      const res = await fetch('/api/wordpress', {
   659→        method: 'POST',
   660→        headers: { 'Content-Type': 'application/json' },
   661→        body: JSON.stringify({
   662→          name: values.name,
   663→          siteUrl: values.siteUrl,
   664→          username: values.username,
   665→          appPassword: values.appPassword,
   666→          postType: values.postType,
   667→          customFields: values.customFields,
   668→        }),
   669→      })
   670→      if (!res.ok) {
   671→        const data = await res.json().catch(() => ({}))
   672→        throw new Error(data?.error || 'Ошибка создания')
   673→      }
   674→      return res.json()
   675→    },
   676→    onSuccess: () => {
   677→      toast.success('Источник добавлен')
   678→      queryClient.invalidateQueries({ queryKey: ['wordpress'] })
   679→      setDialogOpen(false)
   680→    },
   681→    onError: (e: Error) => toast.error(e.message),
   682→  })
   683→
   684→  const updateMutation = useMutation({
   685→    mutationFn: async (vars: { id: string; values: SourceFormValues }) => {
   686→      const res = await fetch(`/api/wordpress/${vars.id}`, {
   687→        method: 'PUT',
   688→        headers: { 'Content-Type': 'application/json' },
   689→        body: JSON.stringify({
   690→          name: vars.values.name,
   691→          siteUrl: vars.values.siteUrl,
   692→          username: vars.values.username,
   693→          appPassword: vars.values.appPassword,
   694→          postType: vars.values.postType,
   695→          customFields: vars.values.customFields,
   696→        }),
   697→      })
   698→      if (!res.ok) {
   699→        const data = await res.json().catch(() => ({}))
   700→        throw new Error(data?.error || 'Ошибка обновления')
   701→      }
   702→      return res.json()
   703→    },
   704→    onSuccess: () => {
   705→      toast.success('Источник обновлён')
   706→      queryClient.invalidateQueries({ queryKey: ['wordpress'] })
   707→      setDialogOpen(false)
   708→      setEditTarget(null)
   709→    },
   710→    onError: (e: Error) => toast.error(e.message),
   711→  })
   712→
   713→  const toggleMutation = useMutation({
   714→    mutationFn: async (vars: { id: string; active: boolean }) => {
   715→      const res = await fetch(`/api/wordpress/${vars.id}`, {
   716→        method: 'PUT',
   717→        headers: { 'Content-Type': 'application/json' },
   718→        body: JSON.stringify({ active: vars.active }),
   719→      })
   720→      if (!res.ok) throw new Error('Не удалось изменить статус')
   721→      return res.json()
   722→    },
   723→    onSuccess: () => {
   724→      queryClient.invalidateQueries({ queryKey: ['wordpress'] })
   725→    },
   726→    onError: (e: Error) => toast.error(e.message),
   727→  })
   728→
   729→  const deleteMutation = useMutation({
   730→    mutationFn: async (id: string) => {
   731→      const res = await fetch(`/api/wordpress/${id}`, { method: 'DELETE' })
   732→      if (!res.ok) throw new Error('Не удалось удалить')
   733→      return res.json()
   734→    },
   735→    onSuccess: () => {
   736→      toast.success('Источник удалён')
   737→      queryClient.invalidateQueries({ queryKey: ['wordpress'] })
   738→      setDeleteTarget(null)
   739→    },
   740→    onError: (e: Error) => toast.error(e.message),
   741→  })
   742→
   743→  const submitting = createMutation.isPending || updateMutation.isPending
   744→
   745→  function openCreate() {
   746→    setEditTarget(null)
   747→    setDialogOpen(true)
   748→  }
   749→
   750→  function openEdit(src: WordPressSource) {
   751→    setEditTarget(src)
   752→    setDialogOpen(true)
   753→  }
   754→
   755→  function handleSubmit(values: SourceFormValues) {
   756→    if (editTarget) {
   757→      updateMutation.mutate({ id: editTarget.id, values })
   758→    } else {
   759→      createMutation.mutate(values)
   760→    }
   761→  }
   762→
   763→  return (
   764→    <div className="grid gap-4">
   765→      <Card className="bg-teal-50/40 dark:bg-teal-950/10">
   766→        <CardHeader>
   767→          <CardTitle className="text-base">WordPress</CardTitle>
   768→          <CardDescription>
   769→            Подключите ваш WordPress-магазин. Посты будут создаваться на основе
   770→            товаров и их кастомных полей. (Полная интеграция с кастомными полями —
   771→            следующий этап.)
   772→          </CardDescription>
   773→        </CardHeader>
   774→      </Card>
   775→
   776→      <div className="flex items-center justify-between gap-2">
   777→        <div className="text-sm text-muted-foreground">
   778→          Источников:{' '}
   779→          <span className="font-medium text-foreground">{data?.length ?? 0}</span>
   780→        </div>
   781→        <Button
   782→          onClick={openCreate}
   783→          className="bg-teal-700 text-white hover:bg-teal-700/90"
   784→        >
   785→          <Plus className="size-4" />
   786→          Добавить источник
   787→        </Button>
   788→      </div>
   789→
   790→      {isLoading ? (
   791→        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   792→          {Array.from({ length: 2 }).map((_, i) => (
   793→            <Skeleton key={i} className="h-72 w-full" />
   794→          ))}
   795→        </div>
   796→      ) : isError ? (
   797→        <Card>
   798→          <CardContent className="py-6 text-sm text-destructive">
   799→            Не удалось загрузить источники.{' '}
   800→            <Button
   801→              variant="link"
   802→              className="h-auto p-0"
   803→              onClick={() => refetch()}
   804→            >
   805→              Повторить
   806→            </Button>
   807→          </CardContent>
   808→        </Card>
   809→      ) : !data || data.length === 0 ? (
   810→        <Card className="border-dashed">
   811→          <CardContent className="flex flex-col items-center gap-3 py-12 text-center">
   812→            <div className="rounded-full bg-muted p-3">
   813→              <Globe className="size-5 text-muted-foreground" />
   814→            </div>
   815→            <div>
   816→              <div className="font-medium">Нет подключённых источников</div>
   817→              <div className="text-sm text-muted-foreground">
   818→                Добавьте WordPress-магазин, чтобы подтягивать товары для постов.
   819→              </div>
   820→            </div>
   821→            <Button
   822→              onClick={openCreate}
   823→              className="bg-teal-700 text-white hover:bg-teal-700/90"
   824→            >
   825→              <Plus className="size-4" />
   826→              Добавить источник
   827→            </Button>
   828→          </CardContent>
   829→        </Card>
   830→      ) : (
   831→        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   832→          {data.map((src) => (
   833→            <SourceCard
   834→              key={src.id}
   835→              source={src}
   836→              onEdit={() => openEdit(src)}
   837→              onDelete={() => setDeleteTarget(src)}
   838→              onShowProducts={() => setProductsSource(src)}
   839→              onToggleActive={(next) =>
   840→                toggleMutation.mutate({ id: src.id, active: next })
   841→              }
   842→              toggling={toggleMutation.isPending}
   843→            />
   844→          ))}
   845→        </div>
   846→      )}
   847→
   848→      <SourceFormDialog
   849→        open={dialogOpen}
   850→        onOpenChange={(o) => {
   851→          setDialogOpen(o)
   852→          if (!o) setEditTarget(null)
   853→        }}
   854→        mode={editTarget ? 'edit' : 'create'}
   855→        initial={editTarget}
   856→        onSubmit={handleSubmit}
   857→        submitting={submitting}
   858→      />
   859→
   860→      <AlertDialog
   861→        open={!!deleteTarget}
   862→        onOpenChange={(o) => !o && setDeleteTarget(null)}
   863→      >
   864→        <AlertDialogContent>
   865→          <AlertDialogHeader>
   866→            <AlertDialogTitle>Удалить источник?</AlertDialogTitle>
   867→            <AlertDialogDescription>
   868→              {deleteTarget?.name ? (
   869→                <>
   870→                  Источник «{deleteTarget.name}» будет удалён без возможности
   871→                  восстановления.
   872→                </>
   873→              ) : (
   874→                'Это действие нельзя отменить.'
   875→              )}
   876→            </AlertDialogDescription>
   877→          </AlertDialogHeader>
   878→          <AlertDialogFooter>
   879→            <AlertDialogCancel>Отмена</AlertDialogCancel>
   880→            <AlertDialogAction
   881→              className="bg-destructive text-white hover:bg-destructive/90"
   882→              disabled={deleteMutation.isPending}
   883→              onClick={(e) => {
   884→                e.preventDefault()
   885→                if (deleteTarget) deleteMutation.mutate(deleteTarget.id)
   886→              }}
   887→            >
   888→              {deleteMutation.isPending ? 'Удаление…' : 'Удалить'}
   889→            </AlertDialogAction>
   890→          </AlertDialogFooter>
   891→        </AlertDialogContent>
   892→      </AlertDialog>
   893→
   894→      <Sheet
   895→        open={!!productsSource}
   896→        onOpenChange={(o) => !o && setProductsSource(null)}
   897→      >
   898→        <SheetContent
   899→          side="right"
   900→          className="w-full flex flex-col gap-4 sm:max-w-xl"
   901→        >
   902→          <SheetHeader>
   903→            <SheetTitle>Товары — {productsSource?.name}</SheetTitle>
   904→            <SheetDescription>
   905→              Загрузка из{' '}
   906→              <span className="font-mono text-xs">
   907→                {productsSource?.siteUrl}
   908→              </span>{' '}
   909→              · тип:{' '}
   910→              <span className="font-mono text-xs">
   911→                {productsSource?.postType || 'product'}
   912→              </span>
   913→            </SheetDescription>
   914→          </SheetHeader>
   915→          <div className="flex-1 overflow-hidden px-4 pb-4">
   916→            {productsSource && (
   917→              <ProductList
   918→                sourceId={productsSource.id}
   919→                customFields={sourceCustomFields(productsSource)}
   920→              />
   921→            )}
   922→          </div>
   923→        </SheetContent>
   924→      </Sheet>
   925→    </div>
   926→  )
   927→}
   928→
   929→export default WordPressSection
