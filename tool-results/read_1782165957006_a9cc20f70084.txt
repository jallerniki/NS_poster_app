   604→   604→export function ButtonsSection() {
   605→   605→  const queryClient = useQueryClient()
   606→   606→
   607→   607→  const { data, isLoading, isError, refetch } = useQuery<ButtonPreset[]>({
   608→   608→    queryKey: ['button-presets'],
   609→   609→    queryFn: async () => {
   610→   610→      const res = await fetch('/api/button-presets')
   611→   611→      if (!res.ok) throw new Error('Не удалось загрузить пресеты кнопок')
   612→   612→      return (await res.json()) as ButtonPreset[]
   613→   613→    },
   614→   614→  })
   615→   615→
   616→   616→  const [dialogOpen, setDialogOpen] = React.useState(false)
   617→   617→  const [editTarget, setEditTarget] = React.useState<ButtonPreset | null>(null)
   618→   618→  const [deleteTarget, setDeleteTarget] = React.useState<ButtonPreset | null>(
   619→   619→    null,
   620→   620→  )
   621→   621→
   622→   622→  const createMutation = useMutation({
   623→   623→    mutationFn: async (values: FormValues) => {
   624→   624→      const res = await fetch('/api/button-presets', {
   625→   625→        method: 'POST',
   626→   626→        headers: { 'Content-Type': 'application/json' },
   627→   627→        body: JSON.stringify(values),
   628→   628→      })
   629→   629→      if (!res.ok) {
   630→   630→        const j = await res.json().catch(() => ({}))
   631→   631→        throw new Error(j?.error || 'Ошибка создания')
   632→   632→      }
   633→   633→      return res.json()
   634→   634→    },
   635→   635→    onSuccess: () => {
   636→   636→      toast.success('Пресет создан')
   637→   637→      queryClient.invalidateQueries({ queryKey: ['button-presets'] })
   638→   638→      setDialogOpen(false)
   639→   639→    },
   640→   640→    onError: (e: Error) => toast.error(e.message),
   641→   641→  })
   642→   642→
   643→   643→  const updateMutation = useMutation({
   644→   644→    mutationFn: async (vars: { id: string; values: FormValues }) => {
   645→   645→      const res = await fetch(`/api/button-presets/${vars.id}`, {
   646→   646→        method: 'PUT',
   647→   647→        headers: { 'Content-Type': 'application/json' },
   648→   648→        body: JSON.stringify(vars.values),
   649→   649→      })
   650→   650→      if (!res.ok) {
   651→   651→        const j = await res.json().catch(() => ({}))
   652→   652→        throw new Error(j?.error || 'Ошибка обновления')
   653→   653→      }
   654→   654→      return res.json()
   655→   655→    },
   656→   656→    onSuccess: () => {
   657→   657→      toast.success('Пресет обновлён')
   658→   658→      queryClient.invalidateQueries({ queryKey: ['button-presets'] })
   659→   659→      setDialogOpen(false)
   660→   660→      setEditTarget(null)
   661→   661→    },
   662→   662→    onError: (e: Error) => toast.error(e.message),
   663→   663→  })
   664→   664→
   665→   665→  const deleteMutation = useMutation({
   666→   666→    mutationFn: async (id: string) => {

   668→   668→        method: 'DELETE',
   669→   669→      })
   670→   670→      if (!res.ok) throw new Error('Не удалось удалить')
   671→   671→      return res.json()
   672→   672→    },
   673→   673→    onSuccess: () => {
   674→   674→      toast.success('Пресет удалён')
   675→   675→      queryClient.invalidateQueries({ queryKey: ['button-presets'] })
   676→   676→      setDeleteTarget(null)
   677→   677→    },
   678→   678→    onError: (e: Error) => toast.error(e.message),
   679→   679→  })
   680→   680→
   681→   681→  const submitting = createMutation.isPending || updateMutation.isPending
   682→   682→
   683→   683→  function openCreate() {
   684→   684→    setEditTarget(null)
   685→   685→    setDialogOpen(true)
   686→   686→  }
   687→   687→
   688→   688→  function openEdit(preset: ButtonPreset) {
   689→   689→    setEditTarget(preset)
   690→   690→    setDialogOpen(true)
   691→   691→  }
   692→   692→
   693→   693→  function handleSubmit(values: FormValues) {
   694→   694→    if (editTarget) {
   695→   695→      updateMutation.mutate({ id: editTarget.id, values })
   696→   696→    } else {
   697→   697→      createMutation.mutate(values)
   698→   698→    }
   699→   699→  }
   700→   700→
   701→   701→  return (
   702→   702→    <div className="grid gap-4">
   703→   703→      <Card className="bg-teal-50/40 dark:bg-teal-950/10">
   704→   704→        <CardHeader>
   705→   705→          <CardTitle className="flex items-center gap-2 text-base">
   706→   706→            <MousePointerClick className="size-4 text-teal-700 dark:text-teal-300" />
   707→   707→            Кнопки в постах
   708→   708→          </CardTitle>
   709→   709→          <CardDescription>
   710→   710→            Пресет кнопок описывает две inline-кнопки — «Товар» и «Категория» —
   711→   711→            которые автоматически добавляются к сообщению при создании поста из
   712→   712→            карточки товара. URL-шаблоны содержат переменные вида{' '}
   713→   713→            <code className="rounded bg-muted px-1 py-0.5 text-[11px]">
   714→   714→              {'{{sku}}'}
   715→   715→            </code>
   716→   716→            , которые подставляются из данных товара WordPress. Создайте по
   717→   717→            одному пресету для каждой площадки (Telegram и MAX).
   718→   718→          </CardDescription>
   719→   719→        </CardHeader>
   720→   720→      </Card>
   721→   721→
   722→   722→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   723→   723→        <div className="text-sm text-muted-foreground">
   724→   724→          {data && data.length > 0
   725→   725→            ? `Всего пресетов: ${data.length}`
   726→   726→            : 'Создайте первый пресет кнопок'}
   727→   727→        </div>
   728→   728→        <Button
   729→   729→          onClick={openCreate}
   730→   730→          className="bg-teal-700 text-white hover:bg-teal-700/90"
   731→   731→        >
   732→   732→          <Plus className="size-4" />
   733→   733→          Новый пресет
   734→   734→        </Button>
   735→   735→      </div>
   736→   736→
   737→   737→      {isLoading ? (
   738→   738→        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   739→   739→          {Array.from({ length: 2 }).map((_, i) => (
   740→   740→            <Skeleton key={i} className="h-56 w-full" />
   741→   741→          ))}
   742→   742→        </div>
   743→   743→      ) : isError ? (
   744→   744→        <Card>
   745→   745→          <CardContent className="py-6 text-sm text-destructive">
   746→   746→            Не удалось загрузить пресеты кнопок.{' '}
   747→   747→            <Button
   748→   748→              variant="link"
   749→   749→              className="h-auto p-0"
   750→   750→              onClick={() => refetch()}
   751→   751→            >
   752→   752→              Повторить
   753→   753→            </Button>
   754→   754→          </CardContent>
   755→   755→        </Card>
   756→   756→      ) : !data || data.length === 0 ? (
   757→   757→        <Card className="border-dashed">
   758→   758→          <CardContent className="flex flex-col items-center gap-3 py-12 text-center">
   759→   759→            <div className="rounded-full bg-muted p-3">
   760→   760→              <MousePointerClick className="size-5 text-muted-foreground" />
   761→   761→            </div>
   762→   762→            <div>
   763→   763→              <div className="font-medium">Пресетов кнопок пока нет</div>
   764→   764→              <div className="text-sm text-muted-foreground">
   765→   765→                Создайте первый пресет — например, кнопки для Telegram-бота
   766→   766→                магазина.
   767→   767→              </div>
   768→   768→            </div>
   769→   769→            <Button
   770→   770→              onClick={openCreate}
   771→   771→              className="bg-teal-700 text-white hover:bg-teal-700/90"
   772→   772→            >
   773→   773→              <Plus className="size-4" />
   774→   774→              Новый пресет
   775→   775→            </Button>
   776→   776→          </CardContent>
   777→   777→        </Card>
   778→   778→      ) : (
   779→   779→        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">

   781→   781→            <ButtonPresetCard

   791→   791→      <ButtonPresetFormDialog

   798→   798→        initial={editTarget}
   799→   799→        onSubmit={handleSubmit}

   803→   803→      <AlertDialog
